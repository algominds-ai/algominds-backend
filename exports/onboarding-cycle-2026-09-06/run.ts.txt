import { spawn } from "node:child_process";
import { readFileSync, appendFileSync, writeFileSync } from "node:fs";

const batch=process.argv[2];
if(!batch || !/^[a-z0-9-]+$/.test(batch))throw new Error("Pass a batch name");
const root="exports/onboarding-cycle-2026-09-06";
const path=`${root}/${batch}-receipts.jsonl`;
const cases=readFileSync(`${root}/${batch}-inputs.json`,"utf8");
const child=spawn("bunx",["vitest","run","--config",".claude/onboarding-cycle/config.ts"],{
  env:{...process.env,SFW_SHIM_DISABLE:"1",ONBOARD_LIVE_FIXTURE:cases},stdio:["ignore","pipe","pipe"],
});
let pending="";
child.stdout.on("data",data=>{
  appendFileSync(`${root}/${batch}-raw.txt`,data);
  pending+=data.toString();
  const lines=pending.split("\n");pending=lines.pop()??"";
  for(const line of lines){
    const at=line.indexOf("RECEIPT ");if(at<0)continue;
    const text=line.slice(at+8);const row=JSON.parse(text);
    appendFileSync(path,text+"\n");
    if(row.status!=="started")writeFileSync(`${root}/${row.id}.json`,JSON.stringify(row,null,2)+"\n");
    console.log(JSON.stringify({id:row.id,status:row.status,dollars:row.cost?.total,seconds:row.seconds}));
  }
});
child.stderr.on("data",data=>appendFileSync(`${root}/${batch}-raw.txt`,data));
const timer=setTimeout(()=>child.kill("SIGINT"),840_000);
child.on("exit",code=>{clearTimeout(timer);process.exitCode=code??1;console.log(`batch=${batch}; exit=${code}`);});
process.on("SIGINT",()=>child.kill("SIGINT"));
