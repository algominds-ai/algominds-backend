import { env as bindings } from "cloudflare:workers";
import { it, expect } from "vitest";
import { z } from "zod";
import { generateStructured, reasoningModel } from "@/core/model";
import { CostLedger } from "@/core/cost";
import { AccountProfile, instructions } from "./shape";

const Case = z.object({id:z.string(),domain:z.string(),note:z.string().nullable(),pages:z.array(z.object({url:z.string(),text:z.string()}))});
const Input = z.object({
  ONBOARD_LIVE_FIXTURE:z.string(), ONBOARD_LIVE_AIG_TOKEN:z.string().min(1),
  ONBOARD_LIVE_GATEWAY_BASE_URL:z.string().min(1), ONBOARD_LIVE_MODEL_ROUTE:z.string().min(1),
});
it("extracts the frozen ICP cases without writing account state",{timeout:840_000},async()=>{
  const input=Input.parse(bindings);
  const cases=z.array(Case).max(12).parse(JSON.parse(input.ONBOARD_LIVE_FIXTURE));
  const env:Env={...bindings,CF_AIG_TOKEN:{get:async()=>input.ONBOARD_LIVE_AIG_TOKEN},AI_GATEWAY_BASE_URL:input.ONBOARD_LIVE_GATEWAY_BASE_URL,MODEL_ROUTE_REASONING:input.ONBOARD_LIVE_MODEL_ROUTE};
  let spent=0;
  for(const test of cases){
    expect(spent).toBeLessThan(2);
    console.log("RECEIPT "+JSON.stringify({id:test.id,status:"started",at:new Date().toISOString()}));
    const ledger=new CostLedger(); const at=Date.now();
    try{
      const output=await generateStructured({model:await reasoningModel(env),configuredId:env.MODEL_ROUTE_REASONING,instructions,prompt:JSON.stringify(test),schema:AccountProfile,headers:{"cf-aig-skip-cache":"true"}},ledger,"onboard-profile-probe");
      expect(output).not.toBeNull();
      spent+=ledger.total();
      console.log("RECEIPT "+JSON.stringify({id:test.id,status:"complete",output,cost:ledger.toJSON(),seconds:(Date.now()-at)/1000}));
    }catch(error){
      console.log("RECEIPT "+JSON.stringify({id:test.id,status:"failed",cost:ledger.toJSON(),seconds:(Date.now()-at)/1000,error:error instanceof Error ? error.message : "unknown error"}));
      throw error;
    }
  }
});
